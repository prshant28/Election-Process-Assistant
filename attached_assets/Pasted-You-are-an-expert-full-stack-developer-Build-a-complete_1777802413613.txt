### 3. API ROUTE (`src/app/api/chat/route.ts`)

Streaming Claude API endpoint:
```typescript
// POST /api/chat
// Body: { messages: Message[], state?: string }
// Returns: streaming text response

import { anthropic } from '@anthropic-ai/sdk'
import { StreamingTextResponse, AnthropicStream } from 'ai'
import { getSystemPrompt } from '@/lib/ai/systemPrompt'
import { getRAGContext } from '@/lib/ai/ragPipeline'
import { isSafeQuery } from '@/lib/ai/safetyFilter'

export async function POST(req: Request) {
  const { messages, state } = await req.json()
  
  // Safety check
  const lastMessage = messages[messages.length - 1].content
  if (!isSafeQuery(lastMessage)) {
    return new Response("I can only help with election process information, not political opinions or candidate recommendations.", { status: 200 })
  }
  
  // RAG: get relevant knowledge chunks
  const context = await getRAGContext(lastMessage)
  
  // Build system prompt with context
  const systemPrompt = getSystemPrompt(context, state)
  
  // Call Claude with streaming
  const response = await anthropic.messages.create({
    model: 'claude-sonnet-4-5',
    max_tokens: 1024,
    system: systemPrompt,
    messages: messages,
    stream: true,
  })
  
  const stream = AnthropicStream(response)
  return new StreamingTextResponse(stream)
}
```

### 6. RAG PIPELINE (`src/lib/ai/ragPipeline.ts`)

Simple cosine-similarity in-memory RAG (no external vector DB):

```typescript
import { electionKnowledge } from '@/lib/data/electionKnowledge'

// Simple keyword + TF-IDF scoring (no external embedding API needed)
export async function getRAGContext(query: string): Promise {
  const queryTerms = query.toLowerCase().split(' ').filter(w => w.length > 3)
  
  const scored = electionKnowledge.map(doc => {
    const text = (doc.title + ' ' + doc.content + ' ' + doc.tags.join(' ')).toLowerCase()
    const score = queryTerms.reduce((acc, term) => {
      const count = (text.match(new RegExp(term, 'g')) || []).length
      return acc + count
    }, 0)
    return { doc, score }
  })
  
  const topDocs = scored
    .sort((a, b) => b.score - a.score)
    .slice(0, 2)
    .filter(d => d.score > 0)
  
  if (topDocs.length === 0) return ''
  
  return topDocs.map(d => 
    `[Source: ${d.doc.title}]\n${d.doc.content}`
  ).join('\n\n---\n\n')
}
```

### 7. SAFETY FILTER (`src/lib/ai/safetyFilter.ts`)

```typescript
const BLOCKED_PATTERNS = [
  /which party (should|to) vote/i,
  /vote for (bjp|congress|aap|sp|bsp|tmc|dmk|aiadmk)/i,
  /is (bjp|congress|aap|modi|rahul|kejriwal) (better|good|bad)/i,
  /(best|worst) (party|candidate|politician)/i,
  /who (will|should) win/i,
  /election (prediction|forecast)/i,
  /opinion on (government|policy|scheme)/i,
]

export function isSafeQuery(message: string): boolean {
  return !BLOCKED_PATTERNS.some(pattern => pattern.test(message))
}
```
